cmake_minimum_required(VERSION 3.20)
project(IkigaiPhysicsTests LANGUAGES CXX)
enable_testing()
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/Physics3D.cmake")
ikigai_add_physics3d_library()
set(IKIGAI_PHYSICS_2D_TESTS ON CACHE BOOL "" FORCE)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/Physics2D.cmake")
add_executable(ikigai_physics_tests physics.cpp)
target_link_libraries(ikigai_physics_tests PRIVATE ikigai_physics3d ikigai_physics2d)
add_test(NAME physics3d_and_fields COMMAND ikigai_physics_tests)
add_executable(ikigai_spatial_tests spatial.cpp)
target_link_libraries(ikigai_spatial_tests PRIVATE ikigai_spatial)
target_compile_definitions(ikigai_spatial_tests PRIVATE USE_PHYSICS_2D=0 USE_PHYSICS_3D=0)
add_test(NAME spatial_utilities COMMAND ikigai_spatial_tests)
add_executable(ikigai_collision_pipeline_tests pipeline.cpp)
target_link_libraries(ikigai_collision_pipeline_tests PRIVATE ikigai_physics2d ikigai_physics3d)
add_test(NAME collision_pipeline COMMAND ikigai_collision_pipeline_tests)
add_executable(ikigai_collision_benchmark benchmark.cpp)
target_link_libraries(ikigai_collision_benchmark PRIVATE ikigai_physics2d ikigai_physics3d)
foreach(enabled 0 1)
    add_executable(ikigai_physics3d_features_${enabled} features.cpp)
    target_compile_features(ikigai_physics3d_features_${enabled} PRIVATE cxx_std_20)
    target_compile_definitions(ikigai_physics3d_features_${enabled} PRIVATE USE_PHYSICS_3D=${enabled})
    target_include_directories(ikigai_physics3d_features_${enabled} PRIVATE
        "${CMAKE_CURRENT_LIST_DIR}/../../../src/engine" "${CMAKE_CURRENT_LIST_DIR}/../../../3rd/json/include")
    add_test(NAME physics3d_features_${enabled} COMMAND ikigai_physics3d_features_${enabled})
endforeach()
foreach(physics2d OFF ON)
    foreach(physics3d OFF ON)
        add_test(NAME physics_flags_${physics2d}_${physics3d} COMMAND "${CMAKE_COMMAND}"
            -S "${CMAKE_CURRENT_LIST_DIR}/flags" -B "${CMAKE_CURRENT_BINARY_DIR}/flags_${physics2d}_${physics3d}"
            -G "${CMAKE_GENERATOR}" -DUSE_PHYSICS_2D=${physics2d} -DUSE_PHYSICS_3D=${physics3d})
    endforeach()
endforeach()
